var ss = SpreadsheetApp.getActiveSpreadsheet(); function doGet(e) { var action = e.parameter.action; // 1. 取得所有月份與各月份的員工名單 if (action === "getInitData") { var sheets = ss.getSheets(); var monthList = []; var monthlyData = {}; sheets.forEach(function(sheet) { var sheetName = sheet.getName(); // 過濾掉不是月份的分頁 (只要工作表名稱包含 "月" 就抓取) if (sheetName.indexOf("月") > -1) { monthList.push(sheetName); // 抓取該月份 B 欄的員工姓名 (從第 2 列開始) var lastRow = sheet.getLastRow(); var names = []; if (lastRow >= 2) { var rawValues = sheet.getRange("B2:B" + lastRow).getValues(); for (var i = 0; i < rawValues.length; i++) { var val = rawValues[i][0]; if (val !== null && val !== undefined && val.toString().trim() !== "") { var cleanName = val.toString().trim(); if (names.indexOf(cleanName) === -1) { // 避免姓名重複 names.push(cleanName); } } } } monthlyData[sheetName] = names; } }); var openMonths = monthList.filter(function(month) { return isMonthOpen_(month); }); return ContentService.createTextOutput(JSON.stringify({ months: monthList, openMonths: openMonths, monthlyData: monthlyData })).setMimeType(ContentService.MimeType.JSON); } // 2. 查詢薪資（直接看網頁選哪個月，就進去那個分頁找人） if (action === "getSalary") { var month = e.parameter.month; var name = e.parameter.name; var password = e.parameter.password; var sheet = ss.getSheetByName(month); // 直接切換到該月份分頁 if (!sheet) return ContentService.createTextOutput(JSON.stringify({status:"error", msg:"找不到該月份"})).setMimeType(ContentService.MimeType.JSON); // 後台可獨立控制每個月份是否開放查詢 if (!isMonthOpen_(month)) { return ContentService.createTextOutput(JSON.stringify({status:"not_open", msg:"該月份薪資尚未開放查詢"})).setMimeType(ContentService.MimeType.JSON); } var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { // B欄姓名(data[i][1])，C欄密碼(data[i][2]) if (data[i][1] && data[i][1].toString().trim() === name && data[i][2] && data[i][2].toString().trim() === password) { var hourlyWage = data[i][3]; // D欄 時薪 var hours = data[i][4]; // E欄 時數 // 判斷是否已開放查詢：如果工時或時薪未填寫（空白），判定為尚未開放 if (hours === "" || hours === null || hours === undefined) { return ContentService.createTextOutput(JSON.stringify({status: "not_open", msg: "該月份薪資尚未開放查詢"})).setMimeType(ContentService.MimeType.JSON); } var result = { hourlyWage: hourlyWage || 0, // D欄 時薪 hours: hours || 0, // E欄 時數 lateMin: data[i][5] || 0, // F欄 遲到(分) advance: data[i][6] || 0, // G欄 預支 loss: data[i][7] || 0, // H欄 營損 totalSalary: data[i][8] || 0, // I欄 實領薪資 missing: data[i][9] || 0, // J欄 缺失次數 lateCount: data[i][10] || 0 // K欄 遲到次數 }; return ContentService.createTextOutput(JSON.stringify({status:"success", data:result})).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({status:"password_error"})).setMimeType(ContentService.MimeType.JSON); } } function isMonthOpen_(month) { var settings = ss.getSheetByName("_後台設定"); if (!settings || settings.getLastRow() < 2) return true; var values = settings.getRange(2, 1, settings.getLastRow() - 1, 2).getValues(); for (var i = 0; i < values.length; i++) { if (String(values[i][0]).trim() === String(month).trim()) { return values[i][1] !== false && String(values[i][1]).toUpperCase() !== "FALSE"; } } return true; } function keepAlive() { // 用來保持伺服器清醒的空指令 Logger.log("Keep alive tick"); }
